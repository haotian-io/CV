\entrytitle
  {CETI: Before the Agent Acts}
  {Apr. 2026 -- Present}

\entryrole{
  First Author
  \textbar{}
  Research under Jialong Li
  \textbar{}
  ZGC Lab
}

\begin{cvitems}
  \item Proposed CETI, a repository-grounded pre-coding review
  workflow that represents task constraints as editable \textbf{Records}
  linked to pinned source passages, allowing developers to revise an
  agent's task interpretation before code generation.

  \item Evaluated with 24 participants across 72 repository-task cases
  (+17.2-point adjusted understanding contrast vs. FreeChat-2) and traced
  request families through 504 coding attempts on 10 tasks and 3 models.
\end{cvitems}

  {Nov. 2025 -- Apr. 2026}
