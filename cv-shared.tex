\cvsection{Research Projects}

\entrytitle
  {CETI: Before the Agent Acts}
  {Mar. 2026 -- Present}

\entryrole{
  First Author
  \textbar{}
  ZGC Lab Intern
  \textbar{}
  Joint work with Jialong Li
}

\begin{cvitems}
  \item Proposed CETI, a repository-grounded pre-coding review
  workflow that represents task constraints as editable \textbf{Records}
  linked to pinned source passages, allowing developers to revise an
  agent's task interpretation before code generation.

  \item Evaluated the workflow with 24 participants across 72
  repository-task cases; the two-cycle Records condition showed the
  largest descriptive understanding gain (+17.2 percentage points
  adjusted contrast vs. FreeChat-2; HC3 95\% CI [--6.0, 40.5]).

  \item Traced 102 confirmed request occurrences through 93
  within-trajectory families, linking 48 behavioral families to patches
  and test evidence across 504 coding attempts on 10 tasks and 3 models.
\end{cvitems}

\entrytitle
  {PilotBench: General Aviation Agent Benchmark}
  {Dec. 2025 -- Mar. 2026}

\entryrole{
  Core Contributor
  \textbar{}
  Remote Research Collaboration
  \textbar{}
  CESS, Beihang University
}

\begin{cvitems}
  \item Co-developed PilotBench from 708 real trajectories,
  34-channel telemetry, and nine flight phases; evaluated 41 models
  with Pilot-Score, a composite of regression accuracy and
  instruction following and constraint compliance.

  \item Identified a precision--controllability dichotomy: traditional
  forecasters reached MAE 7.01, while LLMs achieved 86--89\%
  instruction-following at 11--14 MAE; performance degraded in
  high-workload flight phases.
\end{cvitems}
